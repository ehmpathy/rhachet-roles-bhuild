# criteria.blackbox: a push to a public repo needs an anonymized promise

the one promise literal is `task-to-public-repo-is-anonymized`; below it reads `<promise>`.

# usecase.1 = a push of new text to a public repo
given(a gh.issues target repo whose visibility is public)
  when(the caller pushes a new task with no --promise)
    then(the push exits 2)
    then(no create or edit request reaches the repo)
      sothat(no private detail is public, not even in edit history)
    then(stdout shows the 🦫 nudge: the repo marked `(public)`, the three kinds of detail to check, and the exact command just run with `--promise <promise>` appended)
      sothat(the caller knows what to check and how to proceed, with no doc)
    then(the push leaves no record held for later)
      sothat(a re-run with a scrubbed title leaves no stale twin behind)
  when(the caller re-runs the printed command)
    then(the push exits 0, and exactly one create request reaches the repo)
    then(stdout matches the push output of today, with no nudge)
  when(the caller pushes with --promise and a value other than <promise>)
    then(the push exits 2, names the one valid value, and sends naught)
      sothat(a typo never reads as a silent non-vouch)
  when(the caller pushes with --promise and no value)
    then(the push exits 2, names the one valid value, and sends naught)

# usecase.2 = a push to a private or internal repo
given(a gh.issues target repo whose visibility is private, or internal)
  when(the caller pushes a new task with no --promise)
    then(the push exits 0, and exactly one create request reaches the repo)
    then(stdout matches the push output of today, with no nudge)
      sothat(the common case costs naught new)
  when(the caller pushes with --promise <promise>)
    then(the push exits 0, one request is sent, and the promise has no effect)
  when(the caller updates an issue's description via --exid, with no --promise)
    then(the push exits 0, and exactly one edit request reaches the repo)

# usecase.3 = a push when the visibility cannot be read
given(a gh.issues target repo whose visibility read fails — auth, a token that cannot read it, an absent repo, an answer that does not parse; a rate limit or an outage is a wall instead, see usecase.12)
  when(the caller pushes a new task with no --promise)
    then(the push exits 2, and no create request reaches the repo)
    then(the nudge marks the repo `(visibility unreadable — treated as public)`)
      sothat(a token fault reads apart from a true public repo)
    then(the nudge's last line is the exact command with `--promise <promise>` appended)
  when(the caller pushes with --promise <promise>)
    then(the push proceeds to its send; exit 0 on delivery)
  when(the caller updates an issue's text via --exid, with no --promise)
    then(the push exits 2, the same nudge and marker, the exact --exid command printed, no edit request)

# usecase.4 = an update of an issue's text on a public repo
given(a gh.issues public repo with an extant issue N)
  when(the caller pushes --exid N with a new title or description, no --promise)
    then(the push exits 2, the nudge shown, the exact --exid N command printed)
    then(no edit request reaches the repo)
    then(the push leaves no record held for later)
  when(the same push carries --promise <promise>)
    then(exactly one edit request reaches the repo; exit 0)

# usecase.5 = a status-only update
given(a gh.issues repo of any visibility, and an extant issue N)
  when(the caller pushes --exid N --status CLAIMED, no title, no description, no --promise)
    then(the push exits 0, and the status update reaches the repo)
    then(no nudge is shown)
      sothat(a claim or a delivery on a public repo needs no ceremony)
  when(the same push carries --promise <promise>)
    then(the push exits 0, the promise has no effect)

# usecase.6 = a drain of held records
given(records held while radio was blocked:
      A bound for a public repo, its push made with --promise <promise>;
      B bound for a public repo, its push made with no promise;
      C bound for a private repo;
      D bound for a repo whose visibility cannot be read, no promise)
  when(the caller runs `radio.task.queue drain`)
    then(A and C reach their repos)
    then(B and D do not; each stays held, its reason the public-repo hold)
      sothat(no unvouched text reaches a public repo, and its pusher's task is not lost)
    then(the output names B and D, each with its fix: check its text, then `rhx radio.task.queue drain --promise <promise>`, or drop it)
    then(D's line is marked `(visibility unreadable — treated as public)`)
    then(the drain exits 2)
  when(the caller then runs `radio.task.queue drain --promise <promise>`)
    then(B and D reach their repos; the drain exits 0)
  when(the caller runs `radio.task.queue drain --promise` with any other value)
    then(the drain exits 2, names the one valid value, and sends naught)

# usecase.7 = a push carries out a backlog that holds an unvouched public record
given(a record B held for a public repo with no promise, and the radio gate now open)
  when(the caller pushes a new task to a private repo)
    then(the push's own task reaches its repo; the push exits 0)
    then(B does not reach its repo; it stays held with the public-repo hold as its reason)
    then(the output names B as still held, with `rhx radio.task.queue drain --promise <promise>`)

# usecase.8 = what waits, and help
given(a record held for a public repo with no promise)
  when(the caller runs `radio.task.queue get`)
    then(the record is listed with its reason — the public-repo hold — and its fix)
given(any repo)
  when(the caller runs `radio.task.push --help`)
    then(the help names `--promise <promise>` and says it is owed for text sent to a public repo)
  when(the caller runs `radio.task.queue --help`)
    then(the help names `drain --promise <promise>` and says it vouches for held public records)

# usecase.9 = os.fileops
given(the os.fileops channel)
  when(the caller pushes with or without --promise <promise>)
    then(the push behaves as today; no visibility is read and no nudge is shown)

# usecase.10 = the promise stays bound to its text (catalog #27, #34, #35)
given(a record held for a public repo with a promise)
  when(its body is amended via `radio.task.queue set --description`)
    then(the promise is cleared, and the output says so)
    then(the next drain does not send it without a promise)
  when(the same update is re-pushed with no title and no body, with or without --promise)
    then(no promise is stored; the inherited body is owed, so a public repo nudges)
    then(the nudge names the record file and prints the command with `--description @stdin`)
given(a record held for a public repo with no promise)
  when(its body is scrubbed via `radio.task.queue set --description`, then drained with --promise)
    then(the scrubbed text is sent, never the original)
given(a record that awaits a promise)
  when(the caller runs `radio.task.queue get`, or the stop hook fires)
    then(the reminder names `drain --promise`, never "the next push delivers it")
given(a text+status update on public issue N that was nudged and dropped)
  when(the caller re-runs the push with the status alone)
    then(the status is sent with no nudge; the refused text is not sent)

# usecase.11 = the nudge hides no outcome and strands no update (catalog #28, #29, #33, #37)
given(a public push with no promise)
  when(its drain also carried other records)
    then(beneath the nudge, each other record is named: delivered, or held with its fix)
  when(its record carries text this push did not pass)
    then(the nudge names the record file that holds that text)
  when(updates on the same issue are held behind it)
    then(the nudge names them, and warns when a deliver waits on the dropped claim)
  when(its body holds quotes, `$`, a backtick, or a newline)
    then(the printed command, run as typed, sends the original body byte for byte)

# usecase.12 = order, faults, and empty text (catalog #30, #31, #32, #38)
given(a text-only update on issue N held for its promise)
  when(a later status-only update on N is drained)
    then(the status update is sent; it does not wait behind the text)
given(a visibility read that meets a rate limit or an outage)
  when(the drain reaches it)
    then(the drain halts as for a send wall; the record is not marked unreadable)
given(a visibility read that fails on an auth cause)
  when(the record is held)
    then(its fix leads with `radio.task.queue set --auth as-human`)
given(any push)
  when(its body is empty — an empty `@stdin`, or an empty `--description`)
    then(the push is refused, exit 2; no request is sent)
