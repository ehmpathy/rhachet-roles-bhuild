```ts
/**
 * .what = clamp for the cure on rhachet-roles-ehmpathy's git.stage.add skill
 * .why = the stage gate refused to mark rebase-resolved files unless a separate
 *        stage grant was set. a rebase that only moves commits must ride on the
 *        commit grant alone, and a rebase with no commit grant must still refuse.
 *        this suite runs the INSTALLED skill, so it goes red if the cure is absent
 */
import { spawnSync } from 'child_process';
import * as fs from 'fs';
import * as path from 'path';
import { genTempDir, given, then, useBeforeAll, when } from 'test-fns';

const SKILL_PATH = path.resolve(
  __dirname,
  '../../../node_modules/rhachet-roles-ehmpathy/dist/domain.roles/mechanic/skills/git.stage/git.stage.add.sh',
);

/**
 * .what = run git in a temp repo with a fixed fake identity
 * .why = the temp repo has no git config of its own
 */
const runGit = (input: { cwd: string; args: string[] }) =>
  spawnSync(
    'git',
    [
      '-c',
      'user.name=Test Human',
      '-c',
      'user.email=human@example.com',
      ...input.args,
    ],
    {
      cwd: input.cwd,
      encoding: 'utf-8',
      env: { ...process.env, GIT_EDITOR: 'true' },
    },
  );

/**
 * .what = build a temp repo halted mid-rebase on two conflicts
 * .why = a.txt is a content conflict; d.txt is a modify/delete conflict
 */
const genRepoMidRebase = (input: {
  slug: string;
  meter: { uses: number; push: string; stage: string };
}) => {
  const cwd = genTempDir({ slug: input.slug });

  // base commit on main
  runGit({ cwd, args: ['init', '-b', 'main'] });
  fs.writeFileSync(path.join(cwd, 'a.txt'), 'base\n');
  fs.writeFileSync(path.join(cwd, 'd.txt'), 'base\n');
  runGit({ cwd, args: ['add', '.'] });
  runGit({ cwd, args: ['commit', '-m', 'base'] });

  // feat edits a.txt and deletes d.txt
  runGit({ cwd, args: ['checkout', '-b', 'feat'] });
  fs.writeFileSync(path.join(cwd, 'a.txt'), 'feat\n');
  fs.rmSync(path.join(cwd, 'd.txt'));
  runGit({ cwd, args: ['add', '-A'] });
  runGit({ cwd, args: ['commit', '-m', 'feat'] });

  // main edits both
  runGit({ cwd, args: ['checkout', 'main'] });
  fs.writeFileSync(path.join(cwd, 'a.txt'), 'main\n');
  fs.writeFileSync(path.join(cwd, 'd.txt'), 'main\n');
  runGit({ cwd, args: ['commit', '-am', 'main'] });

  // rebase feat onto main; it halts on both conflicts
  runGit({ cwd, args: ['checkout', 'feat'] });
  runGit({ cwd, args: ['rebase', 'main'] });

  // resolve both in the worktree: a.txt by content, d.txt by deletion
  fs.writeFileSync(path.join(cwd, 'a.txt'), 'resolved\n');
  fs.rmSync(path.join(cwd, 'd.txt'), { force: true });

  // the meter: the grants the mechanic holds
  fs.mkdirSync(path.join(cwd, '.meter'), { recursive: true });
  fs.writeFileSync(
    path.join(cwd, '.meter', 'git.commit.uses.jsonc'),
    JSON.stringify(input.meter),
  );

  return cwd;
};

/**
 * .what = invoke the installed git.stage.add skill
 * .why = HOME points at the temp repo so no global meter leaks in
 */
const invokeStageAdd = (input: { cwd: string; files: string[] }) => {
  const result = spawnSync('bash', [SKILL_PATH, ...input.files], {
    cwd: input.cwd,
    encoding: 'utf-8',
    env: { ...process.env, HOME: input.cwd },
  });
  return {
    exitCode: result.status,
    output: `${result.stdout}${result.stderr}`,
  };
};

/**
 * .what = list the paths git still holds as unmerged
 */
const getUnmerged = (input: { cwd: string }) =>
  runGit({ cwd: input.cwd, args: ['diff', '--name-only', '--diff-filter=U'] })
    .stdout.split('\n')
    .filter(Boolean);

describe('git.stage.add (cured)', () => {
  given(
    '[case1] a rebase halted on a conflict, a commit grant, no stage grant',
    () => {
      const scene = useBeforeAll(async () => {
        const cwd = genRepoMidRebase({
          slug: 'stage-add-rebase-commit-grant',
          meter: { uses: 1, push: 'block', stage: 'block' },
        });
        return { cwd, before: getUnmerged({ cwd }) };
      });

      when('[t0] the mechanic stages the resolved file', () => {
        const result = useBeforeAll(async () =>
          invokeStageAdd({ cwd: scene.cwd, files: ['a.txt'] }),
        );

        then('the conflict was real before the stage', () => {
          expect(scene.before).toContain('a.txt');
        });

        then('it proceeds', () => {
          expect(result.exitCode).toEqual(0);
        });

        then('git no longer holds the file as unmerged', () => {
          expect(getUnmerged({ cwd: scene.cwd })).not.toContain('a.txt');
        });
      });

      when('[t1] the mechanic stages a conflict resolved by deletion', () => {
        const result = useBeforeAll(async () =>
          invokeStageAdd({ cwd: scene.cwd, files: ['d.txt'] }),
        );

        then('it proceeds', () => {
          expect(result.exitCode).toEqual(0);
        });

        then('git no longer holds the deleted path as unmerged', () => {
          expect(getUnmerged({ cwd: scene.cwd })).not.toContain('d.txt');
        });
      });
    },
  );

  given(
    '[case2] a rebase halted on a conflict, NO commit grant, no stage grant',
    () => {
      const scene = useBeforeAll(async () => ({
        cwd: genRepoMidRebase({
          slug: 'stage-add-rebase-no-grant',
          meter: { uses: 0, push: 'block', stage: 'block' },
        }),
      }));

      when('[t0] the mechanic stages the resolved file', () => {
        const result = useBeforeAll(async () =>
          invokeStageAdd({ cwd: scene.cwd, files: ['a.txt'] }),
        );

        then('it is refused as a constraint', () => {
          expect(result.exitCode).toEqual(2);
          expect(result.output).toContain('stage not allowed');
        });

        then('the file stays unmerged', () => {
          expect(getUnmerged({ cwd: scene.cwd })).toContain('a.txt');
        });
      });
    },
  );

  given(
    '[case3] a rebase with a commit grant, and a path outside the conflict set',
    () => {
      const scene = useBeforeAll(async () => {
        const cwd = genRepoMidRebase({
          slug: 'stage-add-rebase-unrelated-path',
          meter: { uses: 1, push: 'block', stage: 'block' },
        });
        fs.writeFileSync(path.join(cwd, 'b.txt'), 'new\n');
        return { cwd };
      });

      when('[t0] the mechanic stages a file no conflict holds', () => {
        const result = useBeforeAll(async () =>
          invokeStageAdd({ cwd: scene.cwd, files: ['b.txt'] }),
        );

        then(
          'it is refused: the commit grant covers only the conflict set',
          () => {
            expect(result.exitCode).toEqual(2);
            expect(result.output).toContain('stage not allowed');
          },
        );
      });
    },
  );
});
```
